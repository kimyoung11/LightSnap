cmake_minimum_required(VERSION 3.20)
project(LightSnapWindows VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(lightsnap-core-tests tests.cpp)
add_test(NAME windows-core COMMAND lightsnap-core-tests)
if(WIN32)
  enable_language(RC)
  add_executable(LightSnap WIN32 main.cpp app.rc)
  target_compile_definitions(LightSnap PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(LightSnap PRIVATE user32 gdi32 shell32 advapi32 comctl32 dwmapi psapi ole32)
  if(MINGW)
    target_link_options(LightSnap PRIVATE -municode -static)
  endif()
  if(MSVC)
    set_property(TARGET LightSnap lightsnap-core-tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(LightSnap PRIVATE /W4 /utf-8 /GS /sdl /guard:cf)
    target_link_options(LightSnap PRIVATE /MANIFEST:NO /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /GUARD:CF /CETCOMPAT)
    target_compile_options(lightsnap-core-tests PRIVATE /W4 /utf-8)
  endif()
  add_test(NAME windows-app-smoke COMMAND LightSnap --self-test)
endif()
